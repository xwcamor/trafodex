Public Sub EnregistrerResultat()

    Dim lRGB As Long
    Dim lngRow As Long
    Dim objRange As Range
    
    Dim strNomSerie As String
    Dim lngNumeroSeriePrecedente As Long
    
    Dim lngSequence As Long
    
    Dim dblCoordonneeX As Double
    Dim dblCoordonneeY As Double
    
    Set objRange = ActiveSheet.Range("rngResultatAnterieur")

    ' Recherche la dernière ligne du range des résultats antérieurs
    lngRow = Right(objRange.Address, Len(objRange.Address) - InStrRev(objRange.Address, "$"))

    ' Construit le nom pour la série
    If objRange.Rows.Count = 2 Then
        lngSequence = 1
    Else
        ' Obtiens la nouvelle séquence
        ' (Fais - 1 pour le max row parce qu'il y a toujours une ligne vide à la fin)
        lngSequence = ActiveSheet.Cells(lngRow - 1, conColNoResultatAnterieur) + 1
    End If

    ' Construit le nom pour la série
    strNomSerie = conNomShapeResultatAnterieur + CStr(lngSequence)

    ' Déverrouiller temporairement la feuille
    DebarrerWS ActiveSheet

    Rows(lngRow).Select

    ' Ajoute une ligne
    Selection.Insert Shift:=xlDown
    
    ' Inscrit les informations du résultat courant
    ActiveSheet.Range("A" & lngRow).Formula = ActiveSheet.Range("rngPourcPPM1")
    ActiveSheet.Range("A" & lngRow).NumberFormat = "0.0"
    
    ActiveSheet.Range("B" & lngRow).Formula = ActiveSheet.Range("rngPourcPPM2")
    ActiveSheet.Range("B" & lngRow).NumberFormat = "0.0"
    
    ActiveSheet.Range("C" & lngRow).Formula = ActiveSheet.Range("rngPourcPPM3")
    ActiveSheet.Range("C" & lngRow).NumberFormat = "0.0"
    
    ActiveSheet.Range("D" & lngRow).Formula = ActiveSheet.Range("rngFault")
    
    ' Recherche la couleur de fond de la cellule de la ligne précédente
    lRGB = GetCouleurShape(lngSequence)
    
    ' Crée un rond de la même couleur dans le tableau des résultats antérieurs
    CreerLegendeCouleurDansResultatAnterieur ActiveSheet.Range("E" & lngRow), lRGB, conNomShapeLegendeResultatAnterieur & lngSequence
    
    
    ' Fais en sorte qu'on peut sélectionner avec la date pour pouvoir la supprimer
    ActiveSheet.Range("F" & lngRow).Locked = False
    ActiveSheet.Range("G" & lngRow).Locked = False
    ActiveSheet.Range("H" & lngRow).Locked = False
    
    ' Merge les 3 prochaines cellules
    Range("F" & lngRow & ":H" & lngRow).Select
    With Selection
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlBottom
        .WrapText = False
        .Orientation = 0
        .AddIndent = False
        .IndentLevel = 0
        .ShrinkToFit = False
        .ReadingOrder = xlContext
        .MergeCells = False
    End With
    
    
    Selection.Merge
    
    ' Inscrit la date et le numéro de la séquence
    ActiveSheet.Range("F" & lngRow).Formula = Now()
    ActiveSheet.Range("K" & lngRow).Formula = lngSequence
    
    ' Fais en sorte qu'on peut sélectionner la séquence pour pouvoir la supprimer
    ActiveSheet.Range("K" & lngRow).Locked = False
    
    ' Get les coordonnées X et Y
    GetCoordonneeResultatActiveSheet dblCoordonneeX, dblCoordonneeY
    
    ' Crée le point dans le graphique
    CreerResultatDansGraphique dblCoordonneeX, dblCoordonneeY, lRGB, strNomSerie

    ' Rebarre la feuille
    BarrerWS ActiveSheet

End Sub

Public Sub SupprimerToutHistoriqueResultat()
' Supprime toutes les lignes dans le range Historique de résultats

    Dim i As Integer
    Dim strNomResultatAnterieur As String
    
    ' Get le nom du range dans la feuille Excel qui représente les résultats antérieurs
    strNomResultatAnterieur = GetNomRangeResultatAnterieur()

    ' Demander une confirmation
    If MsgBox("Êtes-vous certain de vouloir détruire tout l'historique des résultats ? ", vbYesNo + vbQuestion) = vbYes Then
    
        ' Déverrouiller temporairement le calendrier
        DebarrerWS ActiveSheet
    
        ' Passe chacune des lignes de la zone de résultat antérieur
        ' (sauf l'entête et la dernière ligne vide qui ne doit pas être effacée)
        For i = Range(strNomResultatAnterieur).Row + Range(strNomResultatAnterieur).Rows.Count - 2 To Range(strNomResultatAnterieur).Row + 1 Step -1
        
            ' Efface le résultat
            SupprimerUnResultatAnterieur i, False
    
        Next i
        
        ' Rebarre la feuille
        BarrerWS ActiveSheet
        
    End If

End Sub

Public Sub SupprimerUnResultatAnterieur(lRow As Integer, Optional bConfirmerSupprimer As Boolean = True)
' Supprime 1 ligne dans le range des résultats antérieurs
On Error GoTo ErrorHandler

    Dim strNomResultatAnterieur As String
    Dim bCancel As Boolean
    Dim bRebarrerWS As Boolean
    
    ' Get le nom du range dans la feuille Excel qui représente les résultats antérieurs
    strNomResultatAnterieur = GetNomRangeResultatAnterieur()

    ' Vérifier si la cellule courante est dans le range des résultats antérieurs
    If lRow >= Range(strNomResultatAnterieur).Row + 1 And _
       lRow < (Range(strNomResultatAnterieur).Row + Range(strNomResultatAnterieur).Rows.Count) - 1 Then
    
        ' Demander une confirmation si besoin
        If bConfirmerSupprimer Then
            bCancel = MsgBox("Êtes-vous certain de vouloir détruire la ligne # " & lRow & " ? ", vbYesNo + vbQuestion) = vbNo
        End If
    
        ' S'il faut toujours supprimer
        If Not bCancel Then
        
            If IsWSBarre(ActiveSheet) Then
                ' Déverrouiller temporairement le calendrier
                DebarrerWS ActiveSheet
                
                bRebarrerWS = True
            End If

            ' Défait le groupe du triangle
            DefaireGroup
    
            ' Efface la shape associée dans le triangle
            EffacerShape conNomShapeResultatAnterieur & ActiveSheet.Cells(lRow, conColNoResultatAnterieur)
            
            ' Efface la shape legende dans le tableau des résultats
            EffacerShape conNomShapeLegendeResultatAnterieur & ActiveSheet.Cells(lRow, conColNoResultatAnterieur)
    
            ' Efface la ligne courante
            ActiveSheet.Rows(lRow).Delete
        
            ' Regroupe le triangle
            RegrouperShape
            
            If bRebarrerWS Then
                ' Rebarre la feuille
                BarrerWS ActiveSheet
            End If
            
            ' Tout recalculer
            Application.CalculateFull
        End If
    End If
    
    Exit Sub
    
ErrorHandler:
    MsgBox Err.Description
End Sub

Public Function GetNomRangeResultatAnterieur() As String
    ' Retourne le nom du range représentant les résultats antérieurs
    GetNomRangeResultatAnterieur = "rngResultatAnterieur"
End Function

Public Function GetCouleurShape(lngNumero As Long) As Double
    
    Dim lRed As Long
    Dim lGreen As Long
    
    Dim lIntervalleGreen As Long
    Dim lNbIntervalleGreen As Long
    
    lIntervalleGreen = 30
    lNbIntervalleGreen = Fix(255 / lIntervalleGreen) + 1
    
    lRed = 255 - Fix(lngNumero / (lNbIntervalleGreen + 1)) * 30
    lGreen = 255 - (((lngNumero - 1) Mod lNbIntervalleGreen) * lIntervalleGreen)

    GetCouleurShape = RGB(lRed, lGreen, 0)
    
End Function

Private Const conNomFeuilleConfig As String = "Config_Triangle"

Private Const conEchelle As Long = 100
Private Const conNbIntervalle As Long = 10
Private Const conDiametrePoint As Long = 6

Public Sub DessinerTriangle(MyRange As Range)

    ' Coordonnées du triangle
    Dim dblLeft As Double
    Dim dblBottom As Double
    Dim dblWidht As Double
    Dim dblHeight As Double
    
    ' Nombre de colonnes du range
    Dim intNbColonne As Integer
    
    Dim MyShape As Shape

    ' Divise le nombre de cellules par le nombre de lignes pour obtenir
    ' le nombre de colonnes
    intNbColonne = MyRange.Cells.Count / MyRange.Rows.Count

    ' Get les coordonnées qu'aura le triangle selon le range
    dblLeft = MyRange.Cells(MyRange.Rows.Count, 1).Left
    dblWidht = (MyRange.Cells(MyRange.Rows.Count, intNbColonne).Left + MyRange.Cells(MyRange.Rows.Count, intNbColonne).Width) - dblLeft
    dblBottom = MyRange.Cells(MyRange.Rows.Count, 1).Top + MyRange.Cells(MyRange.Rows.Count, 1).Height
    dblHeight = GetHauteurTriangle(dblWidht)

    ' Crée les 3 lignes du triangle
    Set MyShape = ActiveSheet.Shapes.AddLine(dblLeft, dblBottom, dblLeft + dblWidht, dblBottom)
    MyShape.Name = "shpTriangleBottom"

    Set MyShape = ActiveSheet.Shapes.AddLine(dblLeft, dblBottom, dblLeft + (dblWidht / 2), dblBottom - dblHeight)
    MyShape.Name = "shpTriangleLeftSide"

    Set MyShape = ActiveSheet.Shapes.AddLine((dblLeft + dblWidht), dblBottom, dblLeft + (dblWidht / 2), dblBottom - dblHeight)
    MyShape.Name = "shpTriangleRightSide"

    ' Crée les lignes d'intervalles
    CreerLignesIntervalle

    ' Regroupe les élements du triangle
    RegrouperShape

End Sub

Private Sub CreerLignesIntervalle()
    
    Dim i As Integer
    
    Dim x1 As Double
    Dim y1 As Double
    Dim x2 As Double
    Dim y2 As Double
    
    Dim dblValeurIntervalle As Double
    
    Dim MyShape As Shape
    
    ' Calcule la dimension de chaque intervalle
    dblValeurIntervalle = conEchelle / conNbIntervalle
    
    ' Lignes X
    For i = 1 To conNbIntervalle - 1
        
        x1 = GetIntersection(i * dblValeurIntervalle, -1, -1, True)
        y1 = GetIntersection(i * dblValeurIntervalle, -1, -1, False)
        x2 = GetIntersection(i * dblValeurIntervalle, 2, -1, True)
        y2 = GetIntersection(i * dblValeurIntervalle, 2, -1, False)
        
        Set MyShape = ActiveSheet.Shapes.AddLine(GetDeviceCoordonateX(x1), GetDeviceCoordonateY(y1), GetDeviceCoordonateX(x2), GetDeviceCoordonateY(y2))
        MyShape.Name = "shpX" & i
        MyShape.Line.Weight = 1.5

    Next i
    
    ' Lignes Y
    For i = 1 To conNbIntervalle - 1
        
        x1 = GetIntersection(-1, i * dblValeurIntervalle, -1, True)
        y1 = GetIntersection(-1, i * dblValeurIntervalle, -1, False)
        x2 = GetIntersection(-1, i * dblValeurIntervalle, 2, True)
        y2 = GetIntersection(-1, i * dblValeurIntervalle, 2, False)
        
        Set MyShape = ActiveSheet.Shapes.AddLine(GetDeviceCoordonateX(x1), GetDeviceCoordonateY(y1), GetDeviceCoordonateX(x2), GetDeviceCoordonateY(y2))
        MyShape.Name = "shpY" & i
        MyShape.Line.Weight = 1.5
    
    Next i
    
    ' Lignes YY
    For i = 1 To conNbIntervalle - 1
        
        x1 = GetIntersection(-1, -1, i * dblValeurIntervalle, True)
        y1 = GetIntersection(-1, -1, i * dblValeurIntervalle, False)
        x2 = GetIntersection(2, -1, i * dblValeurIntervalle, True)
        y2 = GetIntersection(2, -1, i * dblValeurIntervalle, False)
        
        Set MyShape = ActiveSheet.Shapes.AddLine(GetDeviceCoordonateX(x1), GetDeviceCoordonateY(y1), GetDeviceCoordonateX(x2), GetDeviceCoordonateY(y2))
        MyShape.Name = "shpYY" & i
        MyShape.Line.Weight = 1.5
    
    Next i
    
End Sub

Public Sub CreerLigneZoneFault()

    Dim MyRange As Range

    Dim i As Integer
    
    Dim x1 As Double
    Dim y1 As Double
    Dim x2 As Double
    Dim y2 As Double
    
    Dim MyShape As Shape
    
    ' Défait le groupe du triangle
    DefaireGroup
    
    ' Efface les lignes s'il y en a déjà
    EffacerLigneZoneFault
    
    Set MyRange = ActiveSheet.Range("rngConfigurationZone")
    
    For i = 2 To MyRange.Rows.Count

        If MyRange.Rows.Cells(i - 1, 1).Value = MyRange.Rows.Cells(i, 1).Value Then
    
            x1 = MyRange.Rows.Cells(i - 1, 5).Value
            y1 = MyRange.Rows.Cells(i - 1, 6).Value
    
            x2 = MyRange.Rows.Cells(i, 5).Value
            y2 = MyRange.Rows.Cells(i, 6).Value
    
            Set MyShape = ActiveSheet.Shapes.AddLine(GetDeviceCoordonateX(x1), GetDeviceCoordonateY(y1), GetDeviceCoordonateX(x2), GetDeviceCoordonateY(y2))
            MyShape.Name = conNomShapeLigneZoneFault & i
            MyShape.Line.Weight = 1.5
            
            If MyRange.Rows.Cells(i, 7).Value = 1 Then
                MyShape.Line.DashStyle = msoLineLongDash
            End If
        
        End If
        
    Next i
    
    ' Regroupe les élements du triangle
    RegrouperShape

End Sub

Public Function GetHauteurTriangle(dblWidht As Double) As Double

    ' a = longueur d'un côté
    ' h = altitude du triangle
    '
    '
    '       Racine carrée de 3
    ' h = a ------------------
    '               2
    GetHauteurTriangle = dblWidht * (Math.Sqr(3) / 2)

End Function

Public Function GetDeviceCoordonateX(x As Double) As Double
    GetDeviceCoordonateX = ActiveSheet.Shapes("shpTriangleBottom").Left + (ActiveSheet.Shapes("shpTriangleBottom").Width * (x / 100))
End Function

Public Function GetDeviceCoordonateY(Y As Double) As Double
    GetDeviceCoordonateY = ActiveSheet.Shapes("shpTriangleBottom").Top - ((ActiveSheet.Shapes("shpTriangleBottom").Top - ActiveSheet.Shapes("shpTriangleLeftSide").Top) * (Y / 100))
End Function

Public Function CreerLabel(strLabel As String, x As Long, Y As Long, strName As String, bRecentrerX As Boolean, bRecentrerY As Boolean)

    Dim MyShape As Shape
    
    Set MyShape = ActiveSheet.Shapes.AddLabel(msoTextOrientationHorizontal, x, Y, 0#, 0#)
    
    MyShape.Name = strName
    MyShape.TextFrame.AutoSize = msoTrue
    MyShape.TextFrame.Characters.Text = strLabel
    MyShape.TextFrame.Characters.Font.Bold = True
    
    ' Recentre le X au besoin
    If bRecentrerX Then
        MyShape.IncrementLeft (MyShape.Width / 2) * -1
    End If
    
    ' Recentre le Y au besoin
    If bRecentrerY Then
        MyShape.IncrementTop (MyShape.Height / 2) * -1
    End If

End Function

End Function

Public Sub CreerResultatDansGraphique(x1 As Double, y1 As Double, _
                                      lngCouleur As Long, Optional strName As String, _
                                      Optional bEffacerSiExiste As Boolean = False)
' Crée un point dans le graphique pour le résultat
    
    Dim MyShape As Shape
    
    Dim lDiametre As Long
    Dim bRebarrerWS As Boolean
    
    If IsWSBarre(ActiveSheet) Then
        ' Déverrouiller temporairement la feuille
        DebarrerWS ActiveSheet
        
        bRebarrerWS = True
    End If
    
    ' Défait le groupe du triangle
    DefaireGroup
    
    ' S'il faut effacer la shape si elle existe déjà
    If bEffacerSiExiste Then
        ' Efface la shape si elle existe déjà
        EffacerShape strName
    End If
    
    lDiametre = conDiametrePoint
    
    Set MyShape = ActiveSheet.Shapes.AddShape(msoShapeOval, GetDeviceCoordonateX(x1) - lDiametre / 2, GetDeviceCoordonateY(y1) - lDiametre / 2, lDiametre, lDiametre)
    MyShape.Name = strName
    MyShape.Fill.Solid
    MyShape.Fill.ForeColor.SchemeColor = 0
    MyShape.Line.Visible = msoFalse
    MyShape.Fill.ForeColor.RGB = lngCouleur
    
    ' Regroupe les élements du triangle
    RegrouperShape
    
    ' Rebarre le worksheet si on vient de le débarrer
    If bRebarrerWS Then
        ' Rebarre la feuille
        BarrerWS ActiveSheet
    End If

End Sub

Public Sub CreerLegendeCouleurDansResultatAnterieur(MyRange As Range, _
                                                    lngCouleur As Long, Optional strName As String, _
                                                    Optional bEffacerSiExiste As Boolean = False)
' Crée un point dans le graphique pour le résultat
    
    Dim MyShape As Shape
    
    Dim lDiametre As Long
    
    ' Défait le groupe du triangle
    DefaireGroup
    
    ' S'il faut effacer la shape si elle existe déjà
    If bEffacerSiExiste Then
        ' Efface la shape si elle existe déjà
        EffacerShape strName
    End If
    
    lDiametre = conDiametrePoint
    
    Set MyShape = ActiveSheet.Shapes.AddShape(msoShapeOval, MyRange.Cells(1).Left + (MyRange.Cells(1).Width / 2) - (lDiametre / 2), MyRange.Cells(1).Top + (MyRange.Cells(1).Height / 2) - (lDiametre / 2), lDiametre, lDiametre)

    MyShape.Fill.ForeColor.SchemeColor = 0
    MyShape.Line.Visible = msoFalse

    RegrouperShape

End Sub

Public Sub RegrouperShape()
    
    Dim ArrayShape() As Variant
    
    Dim S As Shape
    Dim i As Integer

    ' Défait le groupe du triangle
    DefaireGroup

    For Each S In ActiveSheet.Shapes
    
        ' Doit commencer par shp
        ' on ne veut pas le groupe complet du triangle
        ' on ne veut pas les cercles de la légende
        If Left(S.Name, 3) = "shp" And S.Name <> conNomShapeGroupTriangle _
                And Left(S.Name, Len(conNomShapeLegendeResultatAnterieur)) <> conNomShapeLegendeResultatAnterieur Then
            ReDim Preserve ArrayShape(i)
            ArrayShape(i) = S.Name
            i = i + 1
        End If

Option Explicit

Public Function GetIntersection(x1 As Double, y1 As Double, yy1 As Double, _
                                bAxis As Boolean) As Double
' Retourne le X ou le Y de l'intersection trouvée
'
' Si bAxis = True
'       Retourne le X
' Sinon
'       Retourne le Y
'
' On doit passer une valeur différente de -1 pour au moins 1 des 3 paramètres (x1, y1, yy1)
'
' Doit passer -1 pour les paramètres dont on ne connait pas les valeurs
'
' Je crée toujours 2 lignes et ensuite je trouve l'intersection entre les 2

    ' Ligne 1
    Dim l1x1 As Double
    Dim l1y1 As Double
    Dim l1x2 As Double
    Dim l1y2 As Double

    ' Ligne 2
    Dim l2x1 As Double
    Dim l2y1 As Double
    Dim l2x2 As Double
    Dim l2y2 As Double

    Dim interX As Double
    Dim interY As Double

    ' Si nous avons le X
    If x1 <> -1 Then
        ' Remplit les informations de la ligne 1
        GetCoordonneesLigne x1, -1, -1, l1x1, l1y1, l1x2, l1y2
    End If
    
    ' Si nous avons le Y mais pas le X
    If y1 <> -1 And x1 = -1 Then
        ' Remplit les informations de la ligne 1
        GetCoordonneesLigne -1, y1, -1, l1x1, l1y1, l1x2, l1y2
        
    ' Si nous avons le Y et le X
    ElseIf y1 <> -1 Then
        ' Remplit les informations de la ligne 2 puisque la ligne 1 est prise par le x
        GetCoordonneesLigne -1, y1, -1, l2x1, l2y1, l2x2, l2y2
    End If
    
    ' Si nous avons le yy mais aucune autre information
    If yy1 <> -1 And x1 = -1 And y1 = -1 Then
        ' Remplit les informations de la ligne 1 puisqu'il n'y a pas de x ni de y
        GetCoordonneesLigne -1, -1, yy1, l1x1, l1y1, l1x2, l1y2
        
    ' si nous avons le yy et 1 des 2 autres informations
    ElseIf yy1 <> -1 And (x1 <> -1 Or y1 <> -1) Then
        ' Remplit les informations de la ligne 2 puisque la ligne 1 est prise par le x ou le y
        GetCoordonneesLigne -1, -1, yy1, l2x1, l2y1, l2x2, l2y2
    End If
    
    ' S'il on a seulement trouvé l'information de 1 ligne (juste un paramètre de passé x1, y1 ou yy1)
    If Abs(x1 <> -1) + Abs(y1 <> -1) + Abs(yy1 <> -1) = 1 Then
    
        ' Si nous avons le X
        If x1 <> -1 Then
            ' Remplit les informations de la ligne 2 en remplissant le y à 0
            GetCoordonneesLigne -1, 0, -1, l2x1, l2y1, l2x2, l2y2
        End If
        
        ' Si nous avons le Y
        If y1 <> -1 Then
            ' Remplit les informations de la ligne 2 en remplissant le yy à 0
            GetCoordonneesLigne -1, -1, 0, l2x1, l2y1, l2x2, l2y2
        End If
        
        ' Si nous avons le YY
        If yy1 <> -1 Then
            ' Remplit les informations de la ligne 2 en remplissant le x à 0
            GetCoordonneesLigne 0, -1, -1, l2x1, l2y1, l2x2, l2y2
        End If
        
    End If
    
    ' Appelle la fonction qui retourne l'intersection
    m_CalculateIntersection l1x1, l1y1, l1x2, l1y2, l2x1, l2y1, l2x2, l2y2, interX, interY
    
    ' Retourne le X ou le Y
    If bAxis Then
        GetIntersection = interX
    Else
        GetIntersection = interY
    End If
    
End Function

Public Sub GetCoordonneesLigne(x1 As Double, y1 As Double, yy1 As Double, _
                                lx1 As Double, ly1 As Double, lx2 As Double, ly2 As Double)
' Retourne le X1, Y1, X2 et Y2 de la ligne

    ' Seulement le X
    If x1 <> -1 And y1 = -1 And yy1 = -1 Then
        lx1 = ConvertirX(x1)
        ly1 = 0
        
        ly2 = lx1
        lx2 = GetXSelonY1(ly2)
    End If
    
    ' Seulement le Y
    If y1 <> -1 And x1 = -1 And yy1 = -1 Then
        lx1 = GetXSelonY1(y1)
        ly1 = y1
        lx2 = 100 - lx1
        ly2 = y1
    End If
    
    ' Seulement le YY (2ième Y)
    If yy1 <> -1 And x1 = -1 And y1 = -1 Then
        lx1 = yy1
        ly1 = 0
        lx2 = GetXSelonY2(yy1)
        ly2 = ConvertirY(yy1)
    End If

End Sub

Public Function GetXSelonY1(y1 As Double) As Double
    ' Divise par 2
    ' (puisque chaque ligne verticale prend la moitié de la ligne horizontale)
    GetXSelonY1 = (y1 / 2)
End Function

Public Function GetXSelonY2(y2 As Double) As Double
    ' Divise par 2 et ajoute 50
    ' (puisque chaque ligne verticale prend la moitié de la ligne horizontale
    '  et qu'on est dans la moitié de droite)
    GetXSelonY2 = GetXSelonY1(y2) + 50
End Function

Public Function ConvertirY(Y As Double) As Double
    ConvertirY = 100 - Y
End Function

Public Function ConvertirX(x As Double) As Double
    ConvertirX = 100 - x
End Function

Public Sub GetCoordonneeResultatActiveSheet(XTrouve As Double, YTrouve As Double)
    XTrouve = GetIntersection(ActiveSheet.Range("rngPourcPPM3"), ActiveSheet.Range("rngPourcPPM1"), -1, True)
    YTrouve = GetIntersection(ActiveSheet.Range("rngPourcPPM3"), ActiveSheet.Range("rngPourcPPM1"), -1, False)
End Sub

Option Explicit

Public Function IntersectComplex(x1 As Double, y1 As Double, x2 As Double, y2 As Double, LineCoordinates As Range, Axis As Boolean) As Variant
'
' Complex Intersect.
' Because the line segments are not uniformly spaced the (xy,y1)(x2,y2) could cross
' at any point along the other line
'
' Return
' If intersection
'    requested coordinate
' else
'    nothing
' endif
' Axis=True returns X value
' Axis=False returns Y value
'
    Dim dblCrossX As Double
    Dim dblCrossY As Double
    Dim dblTestx1 As Double
    Dim dblTesty1 As Double
    Dim dblTestx2 As Double
    Dim dblTesty2 As Double
    Dim intSegment As Integer
    
    With LineCoordinates
        For intSegment = 1 To .Rows.Count - 1
            dblTestx1 = .Cells(intSegment, 1)
            dblTesty1 = .Cells(intSegment, 2)
            dblTestx2 = .Cells(intSegment + 1, 1)
            dblTesty2 = .Cells(intSegment + 1, 2)
            If m_CalculateIntersection(x1, y1, x2, y2, dblTestx1, dblTesty1, dblTestx2, dblTesty2, dblCrossX, dblCrossY) Then
                If Axis Then
                    IntersectComplex = dblCrossX
                Else
                    IntersectComplex = dblCrossY
                End If
                Exit Function
            End If
        Next
    
        ' Special check for last pairing
        intSegment = .Rows.Count
        dblTestx1 = .Cells(intSegment, 1)
        dblTesty1 = .Cells(intSegment, 2)
        dblTestx2 = .Cells(intSegment, 1)
        dblTesty2 = .Cells(intSegment, 2)
        If m_CalculateIntersection(x1, y1, x2, y2, dblTestx1, dblTesty1, dblTestx2, dblTesty2, dblCrossX, dblCrossY) Then
            If Axis Then
                IntersectComplex = dblCrossX
            Else
                IntersectComplex = dblCrossY
            End If
            Exit Function
        End If
        
    End With
    IntersectComplex = CVErr(xlErrNA)    ' Null
    
End Function
Public Function m_CalculateIntersection(x1 As Double, y1 As Double, x2 As Double, y2 As Double, _
    x3 As Double, y3 As Double, x4 As Double, y4 As Double, _
    ByRef CrossX As Double, ByRef CrossY As Double) As Variant

'Call with x1,y1,x2,y2,x3,y3,x4,y4 and returns intersect,x,y
'
'Where:
' x1,y1,x2,y2,x3,y3,x4,y4 are the end points of two line segments
'Returns:
' intersect is true/false, and x,y is the interecting point if intersect is true
'
'Description:
'
'Equations for the lines are:
' Pa = P1 + Ua(P2 - P1)
' Pb = P3 + Ub(P4 - P3)
'
'Solving for the point where Pa = Pb gives the following equations for ua and ub
'
' Ua = ((x4 - x3) * (y1 - y3) - (y4 - y3 ) * (x1 - x3)) / ((y4 - y3) * (x2 - x1)
'     - (x4 - x3) * (y2 - y1))
' Ub = ((x2 - x1) * (y1 - y3) - (y2 - y1 ) * (x1 - x3)) / ((y4 - y3) * (x2 - x1)
'     - (x4 - x3) * (y2 - y1))
'
'Substituting either of these into the corresponding equation for the line gives
'     the intersection point.
'For example the intersection point (x,y) is
' x = x1 + Ua(x2 - x1)
' y = y1 + Ua(y2 - y1)
'
'Notes:
' - The denominators are the same.
'
' - If the denominator above is 0 then the two lines are parallel.
'
' - If the denominator and numerator are 0 then the two lines are coincident.
'
' - The equations above apply to lines,
'     if the intersection of line segments is
'     required then it is only necessary to test if ua and ub lie between 0 and 1.
'     Whichever one lies within that range then the corresponding line segment
'     contains the intersection point. If both lie within the range of 0 to 1 then
'     the intersection point is within both line segments.
'
    Dim dblDenominator As Double
    Dim dblUa As Double
    Dim dblUb As Double
    'Pre calc the denominator, if zero then
    '     both lines are parallel and there is no
    '     intersection
    dblDenominator = ((y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1))

   If dblDenominator <> 0 Then
        'Solve for the simultaneous equations
        dblUa = ((x4 - x3) * (y1 - y3) - (y4 - y3) * (x1 - x3)) / dblDenominator
        dblUb = ((x2 - x1) * (y1 - y3) - (y2 - y1) * (x1 - x3)) / dblDenominator
    Else
    
        If (x1 = x3) And (y1 = y3) Then
            CrossX = x1
            CrossY = y1
            m_CalculateIntersection = True
        ElseIf (x2 = x3) And (y2 = y3) Then
            CrossX = x2
            CrossY = y2
        Else
            m_CalculateIntersection = False
        End If
        Exit Function
    End If
    
' Ancien code
' If dblUa >= 0 And dblUa <= 1 And dblUb >= 0 And dblUb <= dbUn Then
'

    If plusGrandOuEgale(dblUa, CDbl(0)) And plusPetitOuEgale(dblUa, CDbl(1)) And _
       plusGrandOuEgale(dblUb, CDbl(0)) And plusPetitOuEgale(dblUb, CDbl(1)) Then
        'Calculate the intersection point
        CrossX = x1 + dblUa * (x2 - x1)
        CrossY = y1 + dblUa * (y2 - y1)
        'Yes, they do
        m_CalculateIntersection = True
    Else
        'No, they do not
        m_CalculateIntersection = False
    End If
    
End Function

Function plusPetitOuEgale(ByVal val As Double, ByVal ref As Double) As Boolean
Dim dbEpsilon As Double
dbEpsilon = 0.0000000001

If val < ref Then
    plusPetitOuEgale = True
Else
    If (ref - dbEpsilon <= val And val <= ref + dbEpsilon) Then
        plusPetitOuEgale = True
    Else
        plusPetitOuEgale = False
    End If
End If

End Function
Function plusGrandOuEgale(ByVal val As Double, ByVal ref As Double) As Boolean
Dim dbEpsilon As Double
dbEpsilon = 0.0000000001

If val > ref Then
    plusGrandOuEgale = True
Else
    If (ref - dbEpsilon <= val And val <= ref + dbEpsilon) Then
        plusGrandOuEgale = True
    Else
        plusGrandOuEgale = False
    End If
End If

End Function
